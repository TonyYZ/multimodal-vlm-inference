# Matched animation inference: same checkpoints as gestures; animations have no audio.
for model in Qwen/Qwen2.5-Omni-3B Qwen/Qwen2.5-Omni-7B Qwen/Qwen3-Omni-30B-A3B-Instruct; do
  partition=gpu-p1,gpu-p2
  memory=64G
  if [[ "$model" == *30B* ]]; then partition=gpu-p2; memory=128G; fi
  for mode in pure-text split pure-video; do
    sbatch --job-name="anim-${model##*/}-$mode" --output=logs/%x-%j.log --partition="$partition" --gres=gpu:1 --mem="$memory" --time=02:00:00 job_model.sh "$model" --input-mode "$mode"
  done
done

# Temperature sampling: 99 pseudo-participants per statement (T = 1.0, no top-k/top-p), Qwen-VL and Gemma runners only.
for model in Qwen/Qwen2.5-VL-3B-Instruct Qwen/Qwen2.5-VL-7B-Instruct Qwen/Qwen2.5-VL-32B-Instruct Qwen/Qwen3-VL-2B-Instruct Qwen/Qwen3-VL-8B-Instruct Qwen/Qwen3.8-27B google/gemma-4-12B-it; do
  gpus=1; memory=64G; batch=11  # pure-video Qwen3-VL-8B and Qwen3.8-27B need --sample-batch 3 (OOM)
  if [[ "$model" == *32B* || "$model" == *27B* ]]; then gpus=2; memory=128G; batch=9; fi
  for mode in split pure-video; do
    sbatch --job-name="sample-${model##*/}-$mode" --output=logs/%x-%j.log --partition=gpu-p1,gpu-p2,gpu-p3 --gres=gpu:$gpus --mem=$memory --time=08:00:00 job_model.sh "$model" --input-mode "$mode" --samples 99 --temperature 1.0 --sample-batch $batch
  done
done

# Exact rating distributions from logits (robustness check for the temperature samples).
for spec in "Qwen/Qwen2.5-VL-7B-Instruct split" "Qwen/Qwen3-VL-8B-Instruct split" "google/gemma-4-12B-it pure-video"; do
  set -- $spec
  sbatch --job-name="logit-${1##*/}-$2" --output=logs/%x-%j.log --partition=gpu-p1,gpu-p2,gpu-p3 --gres=gpu:1 --mem=64G --time=06:00:00 job_model.sh "$1" --input-mode "$2" --score-distribution
done
